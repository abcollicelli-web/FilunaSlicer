cmake_minimum_required(VERSION 3.16)
project(FilunaNativeContractTest LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenSSL REQUIRED)
find_package(CURL REQUIRED)
add_library(filuna_bambu_adapter STATIC fabricantes/bambu/adapter/BambuAdapter.cpp)
target_include_directories(filuna_bambu_adapter PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/fabricantes)
target_link_libraries(filuna_bambu_adapter PUBLIC OpenSSL::SSL OpenSSL::Crypto CURL::libcurl)
add_executable(filuna_contract_test contract_test.cpp)
target_link_libraries(filuna_contract_test PRIVATE filuna_bambu_adapter)
