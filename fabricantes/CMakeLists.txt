cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(FilunaOEMBatch LANGUAGES CXX)
endif()
# One static target for each OEM. Used by libslic3r_gui AND filuna_batch_test.
option(FILUNA_OEM_BUILD_BATCH_TEST "Build Filuna OEM C++ contract lab" OFF)
option(FILUNA_OEM_LINK_APPLICATION "Link native OEM static targets to Filuna GUI" ON)
find_package(OpenSSL REQUIRED)
find_package(CURL REQUIRED)
add_library(filuna_oem_bambu STATIC bambu/adapter/BambuAdapter.cpp)
target_compile_features(filuna_oem_bambu PUBLIC cxx_std_17)
target_include_directories(filuna_oem_bambu PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(filuna_oem_bambu PUBLIC OpenSSL::SSL OpenSSL::Crypto CURL::libcurl)
if(WIN32)
  target_link_libraries(filuna_oem_bambu PUBLIC ws2_32)
endif()
set(FILUNA_OEM_TARGETS filuna_oem_bambu)
foreach(brand elegoo flashforge qidi creality snapmaker prusa anycubic)
  add_library(filuna_oem_${brand} STATIC ${brand}/adapter/${brand}_adapter.cpp)
  target_compile_features(filuna_oem_${brand} PUBLIC cxx_std_17)
  target_include_directories(filuna_oem_${brand} PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
  list(APPEND FILUNA_OEM_TARGETS filuna_oem_${brand})
endforeach()
if(FILUNA_OEM_LINK_APPLICATION AND TARGET libslic3r_gui)
  target_link_libraries(libslic3r_gui ${FILUNA_OEM_TARGETS})
endif()
if(FILUNA_OEM_BUILD_BATCH_TEST)
  add_executable(filuna_batch_test ${CMAKE_CURRENT_SOURCE_DIR}/../lab/filuna_batch_test.cpp)
  target_link_libraries(filuna_batch_test PRIVATE ${FILUNA_OEM_TARGETS})
  target_include_directories(filuna_batch_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
endif()
